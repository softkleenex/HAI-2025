import os
import cv2
import glob
import torch
from torch.utils.data import Dataset

class DeepFakeDataset(Dataset):
    def __init__(self, root_dir, transform=None, mode='train'):
        self.root_dir = root_dir
        self.transform = transform
        self.mode = mode
        
        # Determine path
        # 140k dataset: /real-vs-fake/{train|valid|test}/{real|fake}
        if mode == 'val': subdir = 'valid'
        elif mode == 'test': subdir = 'test'
        else: subdir = 'train'
        
        target_dir = os.path.join(root_dir, subdir)
        if not os.path.exists(target_dir):
            target_dir = root_dir # Fallback
            
        self.image_paths = glob.glob(os.path.join(target_dir, 'real', '*.jpg')) + \
                           glob.glob(os.path.join(target_dir, 'fake', '*.jpg'))
        self.labels = [0]*len(glob.glob(os.path.join(target_dir, 'real', '*.jpg'))) + \
                      [1]*len(glob.glob(os.path.join(target_dir, 'fake', '*.jpg')))
        print(f"[{mode}] Loaded {len(self.image_paths)} images.")

    def __len__(self): return len(self.image_paths)
    def __getitem__(self, idx):
        try:
            img = cv2.imread(self.image_paths[idx])
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            if self.transform: img = self.transform(image=img)['image']
            return img, self.labels[idx]
        except:
            return torch.zeros((224,224,3)).numpy(), 0
""")

import torch.nn as nn
class DeepFakeClassifier(nn.Module):

        super().__init__()
        self.model = timm.create_model(backbone, pretrained=pretrained, num_classes=num_classes, drop_rate=dropout)
    def forward(self, x): return self.model(x)
""")

# 5. 파일 생성 (train.py - Pseudo Labeling Included)
with open('src/trainer/train.py', 'w') as f:
    f.write("""
import os
import yaml
import argparse
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, ConcatDataset, Dataset
from tqdm import tqdm
import albumentations as A
from albumentations.pytorch import ToTensorV2
from torch.cuda.amp import GradScaler, autocast
from src.models.model import DeepFakeClassifier
from src.data.dataset import DeepFakeDataset
import pandas as pd
import cv2

class PseudoLabelDataset(Dataset):
    def __init__(self, csv_path, transform=None):
        self.data = pd.read_csv(csv_path)
        self.transform = transform
        print(f"Loaded PseudoLabelDataset: {len(self.data)} samples")
    def __len__(self): return len(self.data)
    def __getitem__(self, idx):
        path = self.data.iloc[idx]['path']
        label = int(self.data.iloc[idx]['label'])
        try:
            image = cv2.imread(path)
            if image is None: raise ValueError
            image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
            if self.transform: image = self.transform(image=image)['image']
            return image, label
        except:
            return torch.zeros((224,224,3), dtype=torch.uint8).numpy(), 0

def train(config_path):
    with open(config_path, 'r') as f: config = yaml.safe_load(f)
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"Device: {device}")
    
    img_size = config['data']['img_size']
    train_transform = A.Compose([
        A.Resize(img_size, img_size),
        A.HorizontalFlip(p=0.5),
        A.ImageCompression(quality_lower=60, quality_upper=100, p=0.5),
        A.Normalize(), ToTensorV2()
    ])
    val_transform = A.Compose([A.Resize(img_size, img_size), A.Normalize(), ToTensorV2()])

    train_dataset = DeepFakeDataset(config['data']['train_dir'], transform=train_transform, mode='train')
    
    # Inject Pseudo Labels
    if os.path.exists('data/pseudo_train.csv'):
        print("💉 Injecting Pseudo Labels...")
        pseudo_ds = PseudoLabelDataset('data/pseudo_train.csv', transform=train_transform)
        # Oversampling (x10) logic can be added here if needed
        train_dataset = ConcatDataset([train_dataset, pseudo_ds])
        
    val_dataset = DeepFakeDataset(config['data']['val_dir'], transform=val_transform, mode='val')
    
    train_loader = DataLoader(train_dataset, batch_size=config['data']['batch_size'], shuffle=True, num_workers=2)
    val_loader = DataLoader(val_dataset, batch_size=config['data']['batch_size'], shuffle=False, num_workers=2)

    model = DeepFakeClassifier(backbone=config['model']['backbone'], pretrained=True).to(device)
    optimizer = optim.AdamW(model.parameters(), lr=float(config['train']['lr']))
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=config['train']['epochs'])
    criterion = nn.CrossEntropyLoss()
    scaler = GradScaler()
    
    os.makedirs(config['train']['save_dir'], exist_ok=True)
    best_acc = 0.0

    for epoch in range(config['train']['epochs']):
        model.train()
        train_loss = 0.0
        pbar = tqdm(train_loader, desc=f"Epoch {epoch+1}")
        for images, labels in pbar:
            images, labels = images.to(device), labels.to(device)
            optimizer.zero_grad()
            with autocast():
                outputs = model(images)
                loss = criterion(outputs, labels)
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            train_loss += loss.item()
            pbar.set_postfix({'loss': train_loss / (len(train_loader)+1)})
        
        scheduler.step()
        
        model.eval()
        correct=0; total=0
        with torch.no_grad():
            for images, labels in val_loader:
                images, labels = images.to(device), labels.to(device)
                outputs = model(images)
                _, predicted = torch.max(outputs.data, 1)
                total += labels.size(0)
                correct += (predicted == labels).sum().item()
        acc = 100 * correct / total
        print(f"Val Acc: {acc:.2f}%")
        
        if acc > best_acc:
            best_acc = acc
            torch.save(model.state_dict(), f"{config['train']['save_dir']}/effnet_b5_pseudo_best.pt")
            print("Saved Best Model")

if __name__ == '__main__':
    train('configs/base_config.yaml')
""")
