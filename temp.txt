# 4. Dataset Code
with open('src/data/dataset.py', 'w') as f:
    f.write("""
import os
import cv2
import glob
import torch
from torch.utils.data import Dataset

class DeepFakeDataset(Dataset):
    def __init__(self, root_dir, transform=None, mode='train'):
        self.root_dir = root_dir
        self.transform = transform
        self.mode = mode
        
        subdir = 'valid' if mode == 'val' else ('test' if mode == 'test' else 'train')
        target_dir = os.path.join(root_dir, subdir)
        if not os.path.exists(target_dir): target_dir = root_dir
            
        self.image_paths = glob.glob(os.path.join(target_dir, 'real', '*.jpg')) + \
                           glob.glob(os.path.join(target_dir, 'fake', '*.jpg'))
        self.labels = [0]*len(glob.glob(os.path.join(target_dir, 'real', '*.jpg'))) + \
                      [1]*len(glob.glob(os.path.join(target_dir, 'fake', '*.jpg')))
        print(f"[{mode}] Loaded {len(self.image_paths)} images.")

    def __len__(self): return len(self.image_paths)
    def __getitem__(self, idx):
        try:
            img = cv2.imread(self.image_paths[idx])
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            if self.transform: img = self.transform(image=img)['image']
            return img, self.labels[idx]
        except: return torch.zeros((224,224,3), dtype=torch.uint8).numpy(), 0
""")

# 5. Train Code
with open('src/trainer/train.py', 'w') as f:
    f.write("""
import os
import yaml
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
from tqdm import tqdm
import albumentations as A
from albumentations.pytorch import ToTensorV2
from torch.cuda.amp import GradScaler, autocast
from src.models.model import DeepFakeClassifier
from src.data.dataset import DeepFakeDataset

def train(config_path):
    with open(config_path, 'r') as f: config = yaml.safe_load(f)
    device = torch.device('cuda')
    print(f"Device: {device}")
    
    img_size = config['data']['img_size']
    train_transform = A.Compose([
        A.Resize(img_size, img_size),
        A.HorizontalFlip(p=0.5),
        A.ImageCompression(quality_lower=60, quality_upper=100, p=0.5),
        A.Normalize(), ToTensorV2()
    ])
    val_transform = A.Compose([A.Resize(img_size, img_size), A.Normalize(), ToTensorV2()])

    train_ds = DeepFakeDataset(config['data']['train_dir'], transform=train_transform, mode='train')
    val_ds = DeepFakeDataset(config['data']['val_dir'], transform=val_transform, mode='val')
    
    train_loader = DataLoader(train_ds, batch_size=config['data']['batch_size'], shuffle=True, num_workers=2)
    val_loader = DataLoader(val_ds, batch_size=config['data']['batch_size'], shuffle=False, num_workers=2)

    model = DeepFakeClassifier(backbone=config['model']['backbone']).to(device)
    optimizer = optim.AdamW(model.parameters(), lr=1e-4)
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=config['train']['epochs'])
    criterion = nn.CrossEntropyLoss()
    scaler = GradScaler()
    
    best_acc = 0.0
    for epoch in range(config['train']['epochs']):
        model.train()
        loss_sum = 0
        pbar = tqdm(train_loader, desc=f"Epoch {epoch+1}")
        for imgs, lbls in pbar:
            imgs, lbls = imgs.to(device), lbls.to(device)
            optimizer.zero_grad()
            with autocast():
                out = model(imgs)
                loss = criterion(out, lbls)
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            loss_sum += loss.item()
            pbar.set_postfix({'loss': loss_sum/len(train_loader)})
        
        scheduler.step()
        
        model.eval()
        corr=0; tot=0
        with torch.no_grad():
            for imgs, lbls in val_loader:
                imgs, lbls = imgs.to(device), lbls.to(device)
                out = model(imgs)
                _, pred = torch.max(out, 1)
                tot += lbls.size(0)
                corr += (pred == lbls).sum().item()
        acc = 100 * corr / tot
        print(f"Val Acc: {acc:.2f}%")
        
        if acc > best_acc:
            best_acc = acc
            torch.save(model.state_dict(), f"{config['train']['save_dir']}/convnext_best.pt")
            print("Saved Best Model")

if __name__ == '__main__':
    train('configs/base_config.yaml')
""")
